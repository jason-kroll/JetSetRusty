## Extraction 1 — tile-type mapping (levels.rs) — commit 1

```rust
// Pure mapping cores of level_get_tile_type / level_get_tile_ramp.
pub fn tile_type_for(raw: TileType, air: i32, jump: i32) -> TileType {
    match raw {
        TileType::RampLC => TileType::ConveyR,
        TileType::RampRC => TileType::ConveyL,
        TileType::SolidFloor => {
            if air == 1 && jump == 0 {
                TileType::Solid
            } else {
                TileType::Floor
            }
        }
        t => t,
    }
}

pub fn tile_type_ramp_for(raw: TileType) -> TileType {
    match raw {
        TileType::RampLC => TileType::RampL,
        TileType::RampRC => TileType::RampR,
        t => t,
    }
}

pub fn level_get_tile_ramp(tile: usize) -> TileType {
    if tile >= LEVEL_TILES {
        return TileType::Space;
    }
    tile_type_ramp_for(LEVEL_TILE.with(|t| t.borrow()[tile].tile_type))
}

pub fn level_get_tile_type(tile: usize) -> TileType {
    if tile >= LEVEL_TILES {
        return TileType::Space;
    }
    let raw = LEVEL_TILE.with(|t| t.borrow()[tile].tile_type);
    // Use atomic caches to avoid deadlocking
    let air = GAME_STATE.miner_air.load(Ordering::Relaxed);
    let jump = GAME_STATE.miner_jump.load(Ordering::Relaxed);
    tile_type_for(raw, air, jump)
}
```

## Extraction 2 — is_solid core (miner.rs) — commit 2

```rust
fn is_solid_with(tile: i32, lookup: &dyn Fn(usize) -> TileType, miner: &mut Miner) -> bool {
    if tile < 0 || tile == 512 {
        return false;
    }
    if lookup(tile as usize) == TileType::Solid {
        return true;
    }
    if lookup((tile + 32) as usize) == TileType::Solid {
        return true;
    }
    if tile + 64 > 511 {
        return false;
    }
    if lookup((tile + 64) as usize) != TileType::Solid {
        return false;
    }
    if miner.align == 6 {
        return true;
    }
    if miner.air == 1 && miner.jump > 9 {
        miner.air = 0;
    }
    false
}

fn is_solid(tile: i32, miner: &mut Miner) -> bool {
    let prev_air = miner.air;
    let solid = is_solid_with(tile, &level_get_tile_type, miner);
    if miner.air != prev_air {
        GAME_STATE.miner_air.store(miner.air, Ordering::Relaxed);
    }
    solid
}
```

```rust
enum JumpOutcome {
    Above,
    BumpHead { y: i32, tile: i32 },
    Rise { y: i32, tile: i32, align: i32, pitch: i32, length: i32, apex: bool },
}

fn jump_step(jump_idx: usize, y: i32, tile: i32, lookup: &dyn Fn(usize) -> TileType) -> JumpOutcome {
    let j = JUMP_INFO[jump_idx];
    let ny = y + j.jump;
    if ny < 0 {
        return JumpOutcome::Above;
    }
    let nt = tile + j.tile;
    if lookup(nt as usize) == TileType::Solid || lookup((nt + 1) as usize) == TileType::Solid {
        return JumpOutcome::BumpHead {
            y: (ny + 8) & 120,
            tile: nt + 32,
        };
    }
    JumpOutcome::Rise {
        y: ny,
        tile: nt,
        align: j.align,
        pitch: j.pitch,
        length: j.length,
        apex: jump_idx + 1 == 18,
    }
}
```

## Extraction 4 — ground check (miner.rs) — commit 4

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum GroundOutcome {
    Below,
    HarmPass,
    Die,
    Land { convey_dir: i32 },
    Fall,
}

fn ground_check(tile_base: i32, air: i32, lookup: &dyn Fn(usize) -> TileType) -> GroundOutcome {
    let tile = tile_base + 64;
    if tile & 512 != 0 {
        return GroundOutcome::Below;
    }
    let t0 = lookup(tile as usize);
    let t1 = lookup((tile + 1) as usize);
    if t0 == TileType::Harm || t1 == TileType::Harm {
        if air == 1 && (t0 as i32 <= TileType::Space as i32 || t1 as i32 <= TileType::Space as i32) {
            return GroundOutcome::HarmPass;
        }
        return GroundOutcome::Die;
    }
    if t0 as i32 > TileType::Space as i32 || t1 as i32 > TileType::Space as i32 {
        if air >= 12 {
            return GroundOutcome::Die;
        }
        let convey_dir = if t0 == TileType::ConveyL || t1 == TileType::ConveyL {
            C_LEFT
        } else if t0 == TileType::ConveyR || t1 == TileType::ConveyR {
            C_RIGHT
        } else {
            C_NONE
        };
        return GroundOutcome::Land { convey_dir };
    }
    GroundOutcome::Fall
}
```

## Pitfalls / stop rules

- Pure fns must not read `GAME_STATE`, thread-locals (`LEVEL_TILE`,
  `MINER_FRAME`, …), or input (`system_is_key` etc.) — pass everything in.
  Tests call only the pure fns; never add a test that touches `GAME_STATE`
  (that would race with `game::tests::change_level_above_guard` on parallel
  threads — one such test fn per binary is the existing maximum).
- Keep every `drop(miner)` exactly where it is today; never hold the guard
  across `game_change_level` / `update_dir` / `move_left_right`.
- **Never run `UPDATE_GOLDEN=1` in this plan.** After each commit,
  `git status --short tests/golden` must be empty and `cargo test` fully green
  (unit + all 22 replay). Any divergence = behaviour changed = stop, revert the
  commit, report.
- `cargo clippy --all-targets -- -D warnings` and `cargo fmt --check` must pass
  after every commit (run `cargo fmt` if needed).
- Deliberately NOT extracted in this plan (follow-ups if wanted):
  `move_left_right` (frame animation + edge change-level + rope check) and
  `update_dir` (input reads + rope-jump side effects) — they need an input
  struct / event-enum design; do them as a separate plan.
- Commit style: imperative mood, no `type:` prefix, trailer lines required.
  Do not push.
